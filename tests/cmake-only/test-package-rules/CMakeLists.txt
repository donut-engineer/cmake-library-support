cmake_minimum_required(VERSION 3.15)
project(test-package-rules VERSION 1.0.0 LANGUAGES NONE)

list(APPEND CMAKE_MODULE_PATH "${MODULES_DIR}")
include(cmake-library-support)
include(package-rules)

library_package_rules(
    PACKAGE_NAME "my-library"
    DESCRIPTION  "My Library C++ Library"
)

# include(CPack) inside library_package_rules generates CPackConfig.cmake
if(NOT EXISTS "${PROJECT_BINARY_DIR}/CPackConfig.cmake")
    message(FATAL_ERROR "CPackConfig.cmake was not generated by library_package_rules()")
endif()
